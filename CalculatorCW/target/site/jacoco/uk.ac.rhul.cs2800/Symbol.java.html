package uk.ac.rhul.cs2800;

/**
 * From Dave COHEN, picked from a jar file on Moodle. Enumeration of mathematical symbols used in
 * expressions. Each symbol has a name, ordinal value, and a string representation. Enumeration
 * represents mathematical symbols like +, *, -, and so on.
 *
 * @author abdul
 */
public enum Symbol {
  /**
   * Addition symbol.
   */
  PLUS("PLUS", 0, "+"),
  /**
   * Multiplication symbol.
   */
  MULTI("TIMES", 1, "*"),
  /**
   * Subtraction symbol.
   */
  MINUS("MINUS", 2, "-"),
  /**
   * Unary minus symbol (negation).
   */
  APPROX("SUNIM", 3, "~"),
  /**
   * Division symbol.
   */
  DIVIDE("DIVIDE", 4, "/"),
  /**
   * Exponentiation (power) symbol.
   */
  EXPODIVIDE("EDIVID", 5, "`"),
  /**
   * Left parenthesis symbol.
   */
  LEFT_BRACKET("LEFT_BRACKET", 6, "("),
  /**
   * Right parenthesis symbol.
   */
  RIGHT_BRACKET("RIGHT_BRACKET", 7, ")"),
  /**
   * Invalid or unknown symbol.
   */
  INVALID("INVALID", 8, "?");

  /**
   * The string representation of the symbol.
   */
  private String nm;

  /**
   * Private constructor for the Symbol. Initialize new Symbol with parameters.
   *
   * @param name of the symbol.
   * @param ordinal value of the symbol.
   * @param The string representation of symbol.
   */
  private Symbol(final String name, final int ordinal, final String what) {
    this.nm = what;
  }

  /**
   * Return string representation of symbol.
   *
   * @return string representation of symbol.
   */
  @Override
  public String toString() {
    return new String(this.nm);
  }
}
